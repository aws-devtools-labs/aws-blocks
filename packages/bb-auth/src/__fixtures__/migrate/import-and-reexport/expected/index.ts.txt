import { Auth, AuthErrors, type AuthenticatedUser } from '@aws-blocks/bb-auth';
import { Scope } from '@aws-blocks/core';

export { Auth, AuthErrors } from '@aws-blocks/bb-auth';
export type { AuthenticatedUser } from '@aws-blocks/bb-auth';

const scope = new Scope('app');
export const auth = new Auth(scope, 'auth', { emailPassword: { autoSignIn: false } });
export const notAuthorized = AuthErrors.NotAuthorized;
export type User = AuthenticatedUser;
