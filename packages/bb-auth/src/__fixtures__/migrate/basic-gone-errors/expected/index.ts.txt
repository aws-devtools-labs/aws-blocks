import { Auth } from '@aws-blocks/bb-auth';
import { isBlocksError, Scope } from '@aws-blocks/core';
import type { AuthState } from '@aws-blocks/blocks';

const scope = new Scope('app');
// TODO(aws-blocks-auth-migrate): AuthBasic has no migration path to Auth. Existing users cannot sign in after this deploy and must sign up again.
//   Export them first if you need them (MIGRATION.md, "AuthBasic"); the users / codes tables and the jwt-secret
//   parameter are retained under BlocksPresets.production and deleted under BlocksPresets.sandbox.
//   Sign-up now needs the emailed code (autoSignIn then signs the user in), Cognito sends at most 50 emails/day
//   with its default sender, and Auth's password policy defaults to requiring upper/lower/digit/symbol (the codemod
//   wrote `false` for each one the old policy left unset, so passwords that worked before still do).
export const auth = new Auth(scope, 'auth', { session: { ttlSeconds: 3600 }, emailPassword: { passwordPolicy: { requireUppercase: false, requireLowercase: false, requireDigits: false, requireSymbols: false } } });

// AuthBasic error names with no Auth equivalent: kept, each with a TODO.
export function message(state: AuthState): string {
	// TODO(aws-blocks-auth-migrate): AccountLockedException is gone with AuthBasic's lockout option: Auth never returns it. Remove this check.
	if (state.errorName === 'AccountLockedException') return 'Too many attempts, try again later';
	// TODO(aws-blocks-auth-migrate): SessionRevokedException is gone: Auth reports a signed-out session like a missing one, as NotAuthenticatedException (AuthErrors.NotAuthenticated). Match that, or remove this check.
	if (state.errorName === 'SessionRevokedException') return 'You signed out';
	return '';
}

export function shouldRetrySignOut(e: unknown): boolean {
	// TODO(aws-blocks-auth-migrate): review this auth error check: Auth renamed some error names and changed which flows return which (see MIGRATION.md, "Error names").
	// TODO(aws-blocks-auth-migrate): SignOutIncompleteException is gone: Auth has no error for a sign-out whose server-side revocation failed. Remove this check.
	return isBlocksError(e, 'SignOutIncompleteException');
}
