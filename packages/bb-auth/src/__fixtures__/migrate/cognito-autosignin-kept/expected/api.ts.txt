import type { BlocksContext } from '@aws-blocks/core';
import { auth } from './auth.js';

// A direct signUp() that relied on AuthCognito's default (false) now signs the
// user in: the codemod marks it.
export async function register(username: string, password: string, context: BlocksContext) {
	// TODO(aws-blocks-auth-migrate): this signUp() did not sign the user in (AuthCognito defaulted autoSignIn to false). Auth keeps its default,
	//   emailPassword.autoSignIn: true, because this app also uses createApi() or signUp(…, { autoSignIn: true }), so
	//   confirmSignUp() now returns COMPLETE_AUTO_SIGN_IN instead of DONE here. Handle that step (call autoSignIn()).
	return auth.signUp(username, password, { attributes: { email: username } }, context);
}

// No context: auto sign-in never applied, before or after.
export async function invite(username: string, password: string) {
	return auth.signUp(username, password);
}
