import { Auth } from '@aws-blocks/blocks';
import { Auth as Cognito, AuthErrors } from '@aws-blocks/bb-auth';
import { Scope } from '@aws-blocks/core';

const scope = new Scope('app');
export const users = new Auth(scope, 'users', {});
export const auth = new Cognito(scope, 'auth', { emailPassword: { autoSignIn: false } });
export const code = AuthErrors.NotAuthorized;
