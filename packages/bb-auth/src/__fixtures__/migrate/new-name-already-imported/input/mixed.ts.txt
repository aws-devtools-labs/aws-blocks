import { Auth, AuthErrors } from '@aws-blocks/bb-auth';
import { AuthCognito, AuthCognitoErrors, type CognitoUser } from '@aws-blocks/bb-auth-cognito';
import { Scope } from '@aws-blocks/core';

const scope = new Scope('app');
export const users = new Auth(scope, 'users', {});
export const auth = new AuthCognito(scope, 'auth', {});
export const codes = [AuthErrors.NotAuthorized, AuthCognitoErrors.NotAuthorized];
export type User = CognitoUser;
