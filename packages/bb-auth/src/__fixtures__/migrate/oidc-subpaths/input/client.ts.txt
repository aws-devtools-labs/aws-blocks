import '@aws-blocks/bb-auth-oidc/middleware';
import { AuthOIDCErrors, relayOrigin, type OIDCUser } from '@aws-blocks/bb-auth-oidc/client';
import { AuthOIDCClient, handle401 } from '@aws-blocks/bb-auth-oidc/client';
import { isBlocksError } from '@aws-blocks/core/client';

export const origin = relayOrigin('myapp://callback');
export function name(user: OIDCUser): string {
	return user.username;
}
export function invalid(e: unknown): boolean {
	return isBlocksError(e, AuthOIDCErrors.InvalidState);
}
export function onError(e: unknown): boolean {
	return handle401(e, 'google');
}
export type Client = AuthOIDCClient;
