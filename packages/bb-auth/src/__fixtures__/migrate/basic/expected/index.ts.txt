import { ApiNamespace, Scope } from '@aws-blocks/blocks';
import { Auth, AuthErrors, type AuthenticatedUser } from '@aws-blocks/blocks';

const scope = new Scope('app');
// TODO(aws-blocks-auth-migrate): AuthBasic has no migration path to Auth. Existing users cannot sign in after this deploy and must sign up again.
//   Export them first if you need them (MIGRATION.md, "AuthBasic"); the users / codes tables and the jwt-secret
//   parameter are retained under BlocksPresets.production and deleted under BlocksPresets.sandbox.
//   Sign-up now needs the emailed code (autoSignIn then signs the user in), Cognito sends at most 50 emails/day
//   with its default sender, and Auth's password policy defaults to requiring upper/lower/digit/symbol (the codemod
//   wrote `false` for each one the old policy left unset, so passwords that worked before still do).
const auth = new Auth(scope, 'auth', {
	session: {
		ttlSeconds: 3600,
	},
	emailPassword: {
		passwordPolicy: { minLength: 10, requireSymbols: true, requireUppercase: false, requireLowercase: false, requireDigits: false },
	},
	// TODO(aws-blocks-auth-migrate): codeDelivery is now local-only (npm run dev / tests). On AWS, Cognito emails the codes itself; this hook no longer runs there.
	codeDelivery: async (username, code) => {
		console.log(`code for ${username}: ${code}`);
	},
});

export const authApi = auth.createApi();
// TODO(aws-blocks-auth-migrate): buildApi() is gone: export auth.createApi() and drive it with the Authenticator UI.
export const legacyApi = auth.buildApi();

export const api = new ApiNamespace(scope, 'api', (context) => ({
	async login(username: string, password: string): Promise<AuthenticatedUser> {
		// TODO(aws-blocks-auth-migrate): signIn() now returns { status: "signedIn", user } or { status: "continueSignIn", nextStep } instead of the user.
		const user = await auth.signIn(username, password, context);
		return user;
	},
	async confirm(username: string, code: string) {
		try {
			await auth.confirmSignUp(username, code);
		} catch (e) {
			// TODO(aws-blocks-auth-migrate): InvalidCodeException split in two: a wrong code is now CodeMismatchException (matched here), an expired or missing code is ExpiredCodeException (AuthErrors.ExpiredCode). Add that check if you handled expiry here.
			if (e instanceof Error && e.name === AuthErrors.CodeMismatch) return 'bad code';
			if (e instanceof Error && e.name === AuthErrors.NotAuthorized) return 'denied';
			throw e;
		}
		return 'ok';
	},
}));
