import { hasAuthError } from '@aws-blocks/blocks/client';
import type { AuthState } from '@aws-blocks/blocks';

export function message(state: AuthState): string {
	if (state.errorName === 'NotAuthenticatedException') return 'Please sign in again';
	// TODO(aws-blocks-auth-migrate): review this auth error check: Auth renamed some error names and changed which flows return which (see MIGRATION.md, "Error names").
	// TODO(aws-blocks-auth-migrate): InvalidCodeException split in two: a wrong code is now CodeMismatchException (matched here), an expired or missing code is ExpiredCodeException (AuthErrors.ExpiredCode). Add that check if you handled expiry here.
	if (hasAuthError(state, 'CodeMismatchException')) return 'Wrong code';
	return '';
}
