import { AuthCognito } from '@aws-blocks/bb-auth-cognito';
import { AuthOIDC as Federated, github } from '@aws-blocks/bb-auth-oidc';
import { AppSetting } from '@aws-blocks/bb-app-setting';
import { Scope } from '@aws-blocks/core';

const scope = new Scope('app');
const githubSecret = new AppSetting(scope, 'github-secret', { secret: true });
export const auth = new AuthCognito(scope, 'auth', {});
export const sso = new Federated(scope, 'sso', {
	providers: [github({ clientId: 'gh-client', clientSecret: () => githubSecret.get() })],
});
